%% Main_Cylinder_LNS_NoInterp_Part1.m
% =========================================================================
%  二维可压缩圆柱绕流全局稳定性分析 (原始精度完全版 - Part 1/4)
[cite_start]%  参考文献: Canuto & Taira (JFM 2015)  [cite: 1-628]
%  
%  [功能模块]:
%  1. 参数设置与物理常数定义
%  2. 读取原始 Tecplot 数据 (禁止插值，保持原始精度)
%  3. 输出抽取后的网格数据 (Extracted_Grid_Info.mat)
%  4. 计算全场有限差分导数 (禁止高斯平滑，包含温度二阶导数)
% =========================================================================

clear; 
clc; 
close all;

%% ========================================================================
%  1. 全局参数设置
% ========================================================================
Config.filename = 'cylinder_data.dat';

% --- [原始网格维度设置] ---
% 这里的数值必须与您 cylinder_data.dat 中的实际维度完全一致
% 假设原始数据: i=200 (Wall->Farfield 或反之), j=477 (Periodic)
Config.Nx = 200;  % i (径向)
Config.Ny = 477;  % j (周向, 读取后会进行重合点检查)

Config.N_raw_i = 200; % 原始数据记录 (用于验证)
Config.N_raw_j = 477;

% --- 物理参数 (对应 Canuto 2015 Case) ---
[cite_start]% Re 和 Ma 是影响特征值的关键无量纲参数 [cite: 105-107]
Par.Re    = 50;        % Reynolds Number (Based on Diameter)
Par.Ma    = 0.2;       % Mach Number
Par.gamma = 1.4;       % Ratio of specific heats
Par.Pr    = 0.72;      % Prandtl Number
Par.T_inf = 288.15;    % Free-stream Temperature (K)

% --- 导出热力学参数 ---
% 状态方程: p = rho * R * T (无量纲形式)
Par.Cv = 1 / (Par.gamma * (Par.gamma - 1) * Par.Ma^2);
Par.Cp = Par.gamma * Par.Cv;
Par.R  = 1 / (Par.gamma * Par.Ma^2);

% Sutherland 粘性律常数 (无量纲化)
Par.C_mu = 110.4 / Par.T_inf;

% --- Arnoldi 求解器设置 ---
% 提示: 由于未使用插值，矩阵维度将非常巨大 (4*200*477 ≈ 38万维)
% 内存需求较大，建议使用 32GB+ 内存的工作站
Solver.shift = 0.05 + 0.75i; % 针对 Re=50 圆柱绕流的特征值预估 (St ≈ 0.12)
Solver.k     = 10;           % 求解模态数量
% --- 4th-order semi-artificial viscosity (SAV) ---
% --- 4th-order semi-artificial viscosity (SAV) ---
% The filter is added to the effective operator as
%   L_glob = -A_glob + Kav
% so a negative semi-definite 4th-difference stencil damps compact
% pseudo-modes without changing the physical second-order viscous model.
Config.use_semi_artificial_viscosity = true;
Config.semi_artificial_viscosity = struct( ...
    'epsilon', 5.0e-3, ...
    'use_sensor', false, ...
    'sensor_field', 'pressure', ...
    'sensor_threshold', 0.02, ...
    'sensor_power', 2.0, ...
    'radial_guard_cells', 2);

%% ========================================================================
%  2. 数据读取与预处理 (去除插值与平滑)
% ========================================================================
fprintf('[Step 1] 读取原始 CFD 数据 (无插值模式)...\n');

% 检查文件是否存在
if ~exist(Config.filename, 'file')
    error('错误: 数据文件 %s 未找到。请确保文件路径正确。', Config.filename);
end

% --- [核心修改] 调用 Part 4 定义的 Raw 读取函数 ---
% 该函数将直接 Reshape 原始数据，不进行 interp2 插值
% 注意：Part 4 代码必须与本文件在同一目录下，或合并在同一脚本中
[X, Y, BF, Config] = Read_Raw_Data(Config, Par);

% 更新实际使用的网格维度 (可能因去除周期性重合点而减小 Ny)
Nx = Config.Nx;
Ny = Config.Ny;
fprintf('   [Info] 实际计算网格大小: Nx=%d, Ny=%d (总自由度: %d)\n', ...
        Nx, Ny, 4*Nx*Ny);

% --- 方向一致性检查 ---
% 本代码约定: i=1 (远场 Far Field), i=Nx (壁面 Wall)
% 检查半径 R 分布，如果 i=1 处 R 小于 i=Nx 处 R，说明 i=1 是壁面，需要翻转
R_grid = sqrt(X.^2 + Y.^2);
if mean(R_grid(1,:)) < mean(R_grid(end,:))
    fprintf('   [方向修正] 检测到 i=1 为壁面(小半径)，正在翻转为远场优先 (i=1:Far)...\n');
    X = flipud(X); Y = flipud(Y);
    BF.u = flipud(BF.u); BF.v = flipud(BF.v); 
    BF.T = flipud(BF.T); BF.rho = flipud(BF.rho);
    % 注意: mu 稍后根据 T 重新计算，不需要在此处翻转
end

% --- [边界保护] 强制壁面无滑移 ---
% 假设 i=Nx 为壁面 (根据上述翻转逻辑)
BF.u(end, :) = 0; 
BF.v(end, :) = 0;

%% ========================================================================
%  [用户要求新增] 输出抽取的网格数据
% ========================================================================
fprintf('[Output] 正在输出当前使用的网格数据至文件...\n');
output_file = 'Extracted_Grid_Info.mat';

% 保存 X, Y 坐标以及基本的物理场
% 这里的 X, Y 是经过方向修正和去重合点后的最终网格
save(output_file, 'X', 'Y', 'Nx', 'Ny', 'BF');

fprintf('   -> 网格已保存至: %s\n', output_file);
fprintf('   -> X 维度: %dx%d, Y 维度: %dx%d\n', size(X,1), size(X,2), size(Y,1), size(Y,2));

%% ========================================================================
%  3. 构建差分算子与几何度量 (Step 2)
% ========================================================================
fprintf('[Step 2] 计算全场导数与几何度量 (无平滑模式)... \n');
fprintf('   [警告] 已禁用高斯平滑。二阶导数计算将直接反映原始数据噪声。\n');

% --- 计算粘性及其导数 (Exact Linearization) ---
T_field = BF.T;
C = Par.C_mu;

% mu = T^1.5 * (1+C)/(T+C)
BF.mu = T_field.^1.5 .* (1+C)./(T_field+C);

% dmu/dT
term1 = 1.5./T_field - 1./(T_field+C);
BF.dmudT = BF.mu .* term1;

% d2mu/dT2 (LNS 能量方程源项必须)
term2 = -1.5./(T_field.^2) + 1./((T_field+C).^2);
BF.d2mudT2 = BF.dmudT .* term1 + BF.mu .* term2;

% --- 构建差分算子 ---
% 计算计算空间步长 (用于生成差分矩阵，实际物理距离由 Metric 决定)
dxi = 1/(Nx-1); 
deta = 2*pi/Ny; 

% 获取五点四阶差分矩阵 (函数在 Part 4 定义)
% i方向: 非周期 (Non-periodic)
[d1_xi, d2_xi]   = get_fd_stencil(Nx, 'nonperiodic', dxi);
% j方向: 周期性 (Periodic)
[d1_eta, d2_eta] = get_fd_stencil(Ny, 'periodic', deta);

% Kronecker Product 构建全场算子
% 注意：对于大网格，这里生成的稀疏矩阵依然很大，但 MATLAB 应该能处理
Iy = speye(Ny); Ix = speye(Nx);
D1_xi  = kron(Iy, d1_xi);   
D1_eta = kron(d1_eta, Ix);  
D2_xi  = kron(Iy, d2_xi);   
D2_eta = kron(d2_eta, Ix);  
fprintf('   [Memory] 正在计算几何变换系数 (Metrics)...\n');

% --- 计算几何变换系数 (Metrics) ---
% 使用原始网格坐标计算 Metric，能够自动捕捉网格的拉伸(Stretching)
x_vec = X(:); y_vec = Y(:);

% 物理坐标对计算坐标的导数
x_xi = D1_xi*x_vec; x_eta = D1_eta*x_vec;
y_xi = D1_xi*y_vec; y_eta = D1_eta*y_vec;

J = x_xi.*y_eta - x_eta.*y_xi; % Jacobian

% 检查 Jacobian
if min(abs(J(:))) < 1e-10
    warning('警告: 网格 Jacobian 极小 (<1e-10)，可能存在网格扭曲或重叠。');
end

% 度量系数 (Metrics)
% xi_x = dy/deta / J, etc.
xi_x =  y_eta./J;  eta_x = -y_xi./J;
xi_y = -x_eta./J;  eta_y =  x_xi./J;

% --- 定义全场物理导数计算算子 ---
% 链式法则: d/dx = xi_x * d/dxi + eta_x * d/deta
% 为了节省内存，定义为函数句柄，而不是显式构建大矩阵
calc_dx = @(f) xi_x .* (D1_xi*f) + eta_x .* (D1_eta*f);
calc_dy = @(f) xi_y .* (D1_xi*f) + eta_y .* (D1_eta*f);

% 几何量的二阶导数 (用于粘性项修正)
% 注意: 这里直接对 metric 再次求导
xi_xx = calc_dx(xi_x); xi_yy = calc_dy(xi_y); xi_xy = calc_dx(xi_y);
eta_xx = calc_dx(eta_x); eta_yy = calc_dy(eta_y); eta_xy = calc_dx(eta_y);

% --- [基本流高阶导数计算] ---
% 直接基于原始数据计算，不经过平滑
fprintf('   [Derivatives] 计算基本流一阶与二阶导数...\n');

% 一阶导数
BF.ux = calc_dx(BF.u(:)); BF.uy = calc_dy(BF.u(:));
BF.vx = calc_dx(BF.v(:)); BF.vy = calc_dy(BF.v(:));
BF.Tx = calc_dx(BF.T(:)); BF.Ty = calc_dy(BF.T(:));
BF.rhox = calc_dx(BF.rho(:)); BF.rhoy = calc_dy(BF.rho(:));
BF.mux = calc_dx(BF.mu(:));   BF.muy = calc_dy(BF.mu(:));

% 散度
BF.div_u = BF.ux + BF.vy;

% 计算二阶导数 (源项矩阵 D 需要这些项)
% 注意: 原始数据如果不光滑，这里的 uxx 可能会有显著震荡
BF.uxx = calc_dx(BF.ux); BF.uyy = calc_dy(BF.uy);
BF.vxx = calc_dx(BF.vx); BF.vyy = calc_dy(BF.vy);
BF.uxy = calc_dy(BF.ux); BF.vxy = calc_dy(BF.vx);

% 散度的梯度
BF.div_u_x = calc_dx(BF.div_u);
BF.div_u_y = calc_dy(BF.div_u);

% --- [新增关键项] 温度二阶导数 ---
% 为了在 Part 4 中精确计算 lapT (不进行近似)，这里必须计算 Txx 和 Tyy
fprintf('   [Precision] 计算温度二阶导数 (Txx, Tyy) 以支持精确能量方程源项...\n');
BF.Txx = calc_dx(BF.Tx);
BF.Tyy = calc_dy(BF.Ty);

fprintf('   [完成] Part 1: 数据读取、网格导出与导数计算完毕。\n');
fprintf('   (请继续运行 Part 2 以进行全局矩阵组装)\n');
%% Main_Cylinder_LNS_NoInterp_Part2.m
% =========================================================================
%  二维可压缩圆柱绕流全局稳定性分析 (原始精度完全版 - Part 2/4)
%  功能: 全局矩阵组装 (Global Matrix Assembly)
%  
%  [更新]: 
%  1. 包含了温度二阶导数 (Txx, Tyy) 的传递，用于精确能量方程源项。
%  2. 针对大网格 (200x477) 进行了内存预分配优化。
% =========================================================================

%% ========================================================================
%  4. 全局矩阵组装 (Step 3)
% ========================================================================
fprintf('[Step 3] 组装全局 LNS 矩阵 (维度: %d)...\n', 4*Nx*Ny);

% --- 准备交叉导数算子 ---
% 混合导数项 (Cross-derivatives) 用于粘性应力计算
% D_xieta = D_xi * D_eta
fprintf('   [Pre-calc] 计算交叉导数稀疏算子 D_xieta ...\n');
% 注意: D1_xi 和 D1_eta 在 Part 1 中已构建为全场 Kronecker 形式
D_xieta = D1_xi * D1_eta; 

% --- 内存预分配 ---
N_dim = 4 * Nx * Ny; % 矩阵总维数
% 使用 Triplet 格式 (row, col, val) 构建稀疏矩阵
% 估计非零元: 每个点约关联 5x5=25 个邻居，每个邻居 4x4 块
% 3.8e5 * 25 * 16 ≈ 1.5亿。我们将缓冲区设大一些以防万一。
max_nz = N_dim * 60; 

fprintf('   [Memory] 预分配非零元存储 (Max NZ = %d)...\n', max_nz);
I_A = zeros(max_nz, 1); J_A = zeros(max_nz, 1); V_A = zeros(max_nz, 1);
I_B = zeros(max_nz, 1); J_B = zeros(max_nz, 1); V_B = zeros(max_nz, 1);

count_A = 0;
count_B = 0;

% --- 准备全场数据向量 (加速循环内访问) ---
% 提取 Part 1 计算好的结构体数据
Rho = BF.rho; U = BF.u; V = BF.v; T = BF.T;
Mu  = BF.mu;  DmudT = BF.dmudT; D2mudT2 = BF.d2mudT2;

% 一阶导数
Ux = BF.ux; Uy = BF.uy; Vx = BF.vx; Vy = BF.vy;
Tx = BF.Tx; Ty = BF.Ty; Rhox = BF.rhox; Rhoy = BF.rhoy;
Mux = BF.mux; Muy = BF.muy;

% 二阶导数 (无平滑，包含真实噪声)
Uxx = BF.uxx; Uyy = BF.uyy; Uxy = BF.uxy;
Vxx = BF.vxx; Vyy = BF.vyy; Vxy = BF.vxy;

% [关键新增] 温度二阶导数 (用于精确 lapT)
% 这里的 BF.Txx 和 BF.Tyy 是在 Part 1 结尾计算的
Txx = BF.Txx; 
Tyy = BF.Tyy;

% 散度及其梯度
DivUx = BF.div_u_x; DivUy = BF.div_u_y;

% --- 开始逐点组装 ---
% 这是一个耗时过程，对于 38万个点，可能需要几分钟
fprintf('   正在遍历网格点并组装矩阵...\n');
tic; % 计时开始

for i = 1:Nx
    % 显示进度
    if mod(i, 10) == 0
        fprintf('   -> Progress: Row i = %d / %d (%.1f%%)\n', i, Nx, (i/Nx)*100);
    end
    
    for j = 1:Ny
        g_idx = (j-1)*Nx + i; % 当前节点的全局索引
        
        % -----------------------------------------------------------------
        % (A) 提取当前点的物理量 (封装到结构体 p)
        % -----------------------------------------------------------------
        p.rho = Rho(g_idx);
        p.u = U(g_idx); p.v = V(g_idx); p.T = T(g_idx);
        p.mu = Mu(g_idx);   p.dmudT = DmudT(g_idx); p.d2mudT2 = D2mudT2(g_idx);
        
        p.ux = Ux(g_idx); p.uy = Uy(g_idx);
        p.vx = Vx(g_idx); p.vy = Vy(g_idx);
        p.Tx = Tx(g_idx); p.Ty = Ty(g_idx);
        p.rhox = Rhox(g_idx); p.rhoy = Rhoy(g_idx);
        p.mux = Mux(g_idx); p.muy = Muy(g_idx);
        
        p.uxx = Uxx(g_idx); p.uyy = Uyy(g_idx); p.uxy = Uxy(g_idx);
        p.vxx = Vxx(g_idx); p.vyy = Vyy(g_idx); p.vxy = Vxy(g_idx);
        p.div_u_x = DivUx(g_idx); p.div_u_y = DivUy(g_idx);

        % [关键新增] 传递温度二阶导数
        p.Txx = Txx(g_idx);
        p.Tyy = Tyy(g_idx);

        % -----------------------------------------------------------------
        % (B) 提取当前点的几何度量 (封装到结构体 m)
        % -----------------------------------------------------------------
        m.xi_x = xi_x(g_idx); m.xi_y = xi_y(g_idx);
        m.eta_x = eta_x(g_idx); m.eta_y = eta_y(g_idx);
        
        m.xi_xx = xi_xx(g_idx); m.xi_yy = xi_yy(g_idx); m.xi_xy = xi_xy(g_idx);
        m.eta_xx = eta_xx(g_idx); m.eta_yy = eta_yy(g_idx); m.eta_xy = eta_xy(g_idx);

        % -----------------------------------------------------------------
        % (C) 调用核心系数函数 (Nxishujuzhen)
        % -----------------------------------------------------------------
        % 该函数在 Part 4 定义
        [Gamma, At, Bt, Dt, V1, V2, V12] = Nxishujuzhen(p, m, Par);

        % -----------------------------------------------------------------
        % (D) 填充全局矩阵 B (Mass Matrix)
        % -----------------------------------------------------------------
        rows = (g_idx-1)*4 + (1:4); % 当前点对应的 4 个方程行号
        
        for r = 1:4
            count_B = count_B + 1;
            I_B(count_B) = rows(r);
            J_B(count_B) = rows(r);
            V_B(count_B) = Gamma(r,r);
        end
        
        % -----------------------------------------------------------------
        % (E) 填充全局矩阵 A (Jacobian Operator)
        % -----------------------------------------------------------------
        % 形式: dq/dt = -inv(B) * A * q
        % 这里构建的是算子 A_glob (Equation Left-Hand Side when moved: -A*q)
        
        % 1. 源项 Dt (Local Block, 4x4)
        % -----------------------------
        % Dt 仅与当前节点 q(i,j) 有关
        [Ir, Jr, Vr] = find(Dt);
        for k = 1:length(Vr)
            count_A = count_A + 1;
            I_A(count_A) = rows(Ir(k));
            J_A(count_A) = rows(Jr(k));
            V_A(count_A) = Vr(k);
        end
        
        % 2. ξ 方向对流项: At * D1_xi
        % -----------------------------
        [~, c_xi, v_xi] = find(D1_xi(g_idx, :));
        % 提取 At 的非零元以减少内层循环次数
        [Ir_At, Jr_At, Vr_At] = find(At);
        
        for k = 1:length(c_xi) 
            target_col_block = (c_xi(k)-1)*4 + (1:4);
            for n = 1:length(Vr_At)
                count_A = count_A + 1;
                I_A(count_A) = rows(Ir_At(n));
                J_A(count_A) = target_col_block(Jr_At(n));
                V_A(count_A) = Vr_At(n) * v_xi(k);
            end
        end
        
        % 3. η 方向对流项: Bt * D1_eta
        % -----------------------------
        [~, c_eta, v_eta] = find(D1_eta(g_idx, :));
        [Ir_Bt, Jr_Bt, Vr_Bt] = find(Bt);
        
        for k = 1:length(c_eta)
            target_col_block = (c_eta(k)-1)*4 + (1:4);
            for n = 1:length(Vr_Bt)
                count_A = count_A + 1;
                I_A(count_A) = rows(Ir_Bt(n));
                J_A(count_A) = target_col_block(Jr_Bt(n));
                V_A(count_A) = Vr_Bt(n) * v_eta(k);
            end
        end
        
        % 4. ξ 方向粘性项: -V1 * D2_xi
        % -----------------------------
        [~, c_xi2, v_xi2] = find(D2_xi(g_idx, :));
        [Ir_V1, Jr_V1, Vr_V1] = find(V1);
        
        for k = 1:length(c_xi2)
            target_col_block = (c_xi2(k)-1)*4 + (1:4);
            for n = 1:length(Vr_V1)
                count_A = count_A + 1;
                I_A(count_A) = rows(Ir_V1(n));
                J_A(count_A) = target_col_block(Jr_V1(n));
                V_A(count_A) = -Vr_V1(n) * v_xi2(k); % 注意负号(移项)
            end
        end
        
        % 5. η 方向粘性项: -V2 * D2_eta
        % -----------------------------
        [~, c_eta2, v_eta2] = find(D2_eta(g_idx, :));
        [Ir_V2, Jr_V2, Vr_V2] = find(V2);
        
        for k = 1:length(c_eta2)
            target_col_block = (c_eta2(k)-1)*4 + (1:4);
            for n = 1:length(Vr_V2)
                count_A = count_A + 1;
                I_A(count_A) = rows(Ir_V2(n));
                J_A(count_A) = target_col_block(Jr_V2(n));
                V_A(count_A) = -Vr_V2(n) * v_eta2(k);
            end
        end
        
        % 6. 混合导数项: -V12 * D_xieta
        % -----------------------------
        [~, c_xieta, v_xieta] = find(D_xieta(g_idx, :));
        [Ir_V12, Jr_V12, Vr_V12] = find(V12);
        
        for k = 1:length(c_xieta)
            target_col_block = (c_xieta(k)-1)*4 + (1:4);
            for n = 1:length(Vr_V12)
                count_A = count_A + 1;
                I_A(count_A) = rows(Ir_V12(n));
                J_A(count_A) = target_col_block(Jr_V12(n));
                V_A(count_A) = -Vr_V12(n) * v_xieta(k);
            end
        end
        
    end
end
assembly_time = toc;
fprintf('   [Time] 矩阵组装耗时: %.2f 秒。\n', assembly_time);

% --- 构建稀疏矩阵对象 ---
fprintf('   正在构建压缩稀疏矩阵 (Sparse Construction)...\n');
% 截断多余的零元预分配
I_A = I_A(1:count_A); J_A = J_A(1:count_A); V_A = V_A(1:count_A);
I_B = I_B(1:count_B); J_B = J_B(1:count_B); V_B = V_B(1:count_B);

A_glob = sparse(I_A, J_A, V_A, N_dim, N_dim);
B_glob = sparse(I_B, J_B, V_B, N_dim, N_dim);
% --- 4th-order semi-artificial viscosity on the effective operator ---
[Kav, SavInfo] = build_cylinder_semi_artificial_viscosity( ...
    Nx, Ny, BF, Par, Config, calc_dx, calc_dy);

    fprintf(['   [SAV] Mode=%s, epsilon=%.3e, active rows=%d, ', ...

else

%% Main_Cylinder_LNS_NoInterp_Part3.m
% =========================================================================
%  二维可压缩圆柱绕流全局稳定性分析 (原始精度完全版 - Part 3/4)
%  功能: 边界条件施加、大规模特征值求解、结果可视化
%  
%  [特征]:
%  1. 使用 4 阶偏心差分施加壁面密度边界条件 (No Approximation)。
%  2. 针对 38万维矩阵的 Arnoldi 求解器设置。
% =========================================================================

%% ========================================================================
%  5. 施加边界条件 (High-Order Boundary Closure)
% ========================================================================
fprintf('[Step 4] 施加边界条件 (i=1:Farfield, i=%d:Wall)...\n', Nx);

% --- 索引定义 ---
% 网格存储顺序: i (径向) 变化快, j (周向) 变化慢
% g_idx = (j-1)*Nx + i
% 远场边界 (i=1): Index = 1, 1+Nx, 1+2Nx...
idx_far = 1:Nx:Nx*Ny;
% 壁面边界 (i=Nx): Index = Nx, 2Nx, 3Nx...
idx_wall = Nx:Nx:Nx*Ny;

% -------------------------------------------------------------------------
% (1) 远场边界 (Far Field, i=1) -> Dirichlet Condition
% -------------------------------------------------------------------------
% 物理意义: 扰动在无穷远处衰减为 0
% 实现: rho'=0, u'=0, v'=0, T'=0
for k = idx_far
    for v = 1:4 
        r = (k-1)*4 + v; % 方程行索引
        
        % 清空该行所有元素 (A 和 B)
        % 在稀疏矩阵中，虽然置零操作较慢，但对于边界点数量级是可以接受的
        A_glob(r, :) = 0;
        B_glob(r, :) = 0; 
        
        % 对角置 1，右端项为 0 => 1*dq/dt = -1*q => q(t) -> 0
        % 或者理解为: 1*q = 0 (代数约束)
        A_glob(r, r) = 1; 
    end
end

% -------------------------------------------------------------------------
% (2) 壁面边界 (Wall, i=Nx) -> No-Slip + Compatibility
% -------------------------------------------------------------------------
for k = idx_wall
    % -- A. 速度与温度 (Dirichlet) --
    % u'=0, v'=0 (无滑移), T'=0 (假设等温壁面/无温度扰动)
    for v = 2:4 
        r = (k-1)*4 + v;
        A_glob(r, :) = 0; 
        B_glob(r, :) = 0; 
        A_glob(r, r) = 1;
    end
    
    % -- B. 密度 (相容性条件/高阶 Neumann) --
    % [重要] 既然内部使用了高阶差分，边界必须匹配以防止数值反射。
    % 物理含义: d(rho)/dn = 0 (边界层近似) 或 dP/dn = 0
    % 使用 "5点向后偏心差分" (Backward 4th order) 对应 i=Nx
    
    r = (k-1)*4 + 1; % 密度方程所在的行
    A_glob(r, :) = 0; 
    B_glob(r, :) = 0;
    
    % 计算相邻节点的列索引 (注意变量偏移量 +0 代表 rho)
    col_N   = (k-1)*4 + 1; % i=Nx
    col_Nm1 = (k-2)*4 + 1; % i=Nx-1
    col_Nm2 = (k-3)*4 + 1; % i=Nx-2
    col_Nm3 = (k-4)*4 + 1; % i=Nx-3
    col_Nm4 = (k-5)*4 + 1; % i=Nx-4
    
    % 填入差分系数 (Explicit coefficients for 4th order biased stencil)
    % 25*f(N) - 48*f(N-1) + 36*f(N-2) - 16*f(N-3) + 3*f(N-4) = 0
    A_glob(r, col_N)   = 25;
    A_glob(r, col_Nm1) = -48;
    A_glob(r, col_Nm2) = 36;
    A_glob(r, col_Nm3) = -16;
    A_glob(r, col_Nm4) = 3;
end

%% ========================================================================
%  6. 特征值求解 (Arnoldi Solver)
% ========================================================================
% LNS方程: B * dq/dt = -A * q  =>  lambda * B * q = -A * q
% MATLAB eigs 求解: A_mat * x = lambda * B_mat * x
% 因此我们传入 -A_glob 和 B_glob

fprintf('[Step 5] 开始求解特征值 (大规模矩阵求解, 请耐心等待)...\n');
fprintf('   Target Shift: %.2f+%.2fi\n', real(Solver.shift), imag(Solver.shift));
fprintf('   Matrix Size : %d x %d\n', 4*Nx*Ny, 4*Nx*Ny);

% Arnoldi 迭代参数优化
opts.isreal = 0; 
opts.p = 40;      % Arnoldi 向量数 (增加此值可提高收敛性，但消耗内存)
opts.tol = 1e-8;  % 收敛容差 (1e-8 足够用于流体稳定性)
opts.maxit = 300; % 最大迭代次数
L_glob = -A_glob;
L_glob = -A_glob;
if exist('Kav', 'var') && ~isempty(Kav)
    L_glob = L_glob + Kav;
end

try
    tic;
    % [Solver Call]
    % Effective operator already includes SAV through L_glob = -A_glob + Kav
    % 注意负号: -A_glob
    [V_res, D_res] = eigs(L_glob, B_glob, Solver.k, Solver.shift, opts);
    solve_time = toc;
    fprintf('   求解完成，耗时 %.2f 秒。\n', solve_time);
    
    evals = diag(D_res);
    % 按实部(增长率)从大到小排序，实部最大者为最不稳定模态
    [~, idx] = sort(real(evals), 'descend');
    
    best_lam = evals(idx(1));
    best_vec = V_res(:, idx(1));
    
    fprintf('   ------------------------------------------------\n');
    fprintf('   最不稳定模态 (Most Unstable Mode):\n');
    fprintf('   Eigenvalue = %.5f %+.5fi\n', real(best_lam), imag(best_lam));
    fprintf('   Strouhal Number (St) ≈ %.4f\n', abs(imag(best_lam))/(2*pi));
    fprintf('   Growth Rate (sigma)  = %.5f\n', real(best_lam));
    fprintf('   ------------------------------------------------\n');
    
catch ME
    fprintf('\n[Error] Arnoldi 求解失败: %s\n', ME.message);
    fprintf('建议尝试:\n');
    fprintf('   1. 减小 Solver.k (当前=%d)\n', Solver.k);
    fprintf('   2. 减小 opts.p (当前=%d)\n', opts.p);
    fprintf('   3. 检查是否有 NaN/Inf 进入矩阵 (Part 1/2)。\n');
    return;
end

%% ========================================================================
%  7. 结果绘图与后处理
% ========================================================================
fprintf('[Step 6] 绘制结果云图...\n');

% --- 重构特征向量 (1D向量 -> 2D物理场) ---

rho_hat = reshape(best_vec(1:4:end), Nx, Ny);
u_hat   = reshape(best_vec(2:4:end), Nx, Ny);
v_hat   = reshape(best_vec(3:4:end), Nx, Ny);

p_hat = (1/gM2) * (BF.rho .* T_hat + BF.T .* rho_hat);

% --- 归一化 (Normalization) ---
% 通常以最大 u 速度扰动的模长为基准进行归一化
norm_fac = max(abs(u_hat(:)));
if norm_fac == 0, norm_fac = 1; end

% 准备绘图数据 (取实部 Real Part 用于可视化瞬时快照)
plot_vars = {rho_hat, u_hat, v_hat, T_hat, p_hat};
titles = {'Density (\rho'')', 'Streamwise Vel (u'')', 'Transverse Vel (v'')', ...

figure('Color','w','Name','Global Stability Results (Raw Grid)','Position',[100,100,1600,900]);

% 1. 绘制特征值谱 (Spectrum)
subplot(2,3,1); 
plot(real(evals), imag(evals), 'bo', 'MarkerFaceColor', 'b', 'MarkerSize', 6);
hold on;

grid on;

%% Main_Cylinder_LNS_NoInterp_Part4.m
% =========================================================================
%  二维可压缩圆柱绕流全局稳定性分析 (原始精度完全版 - Part 4/4)
%  功能: 核心物理函数 (精确源项)、高阶差分生成、原始数据读取
%  
%  [特征]:
%  1. Nxishujuzhen: 包含精确的能量方程源项 (No approximation on lapT).
%  2. Read_Raw_Data: 纯粹的数据重构，无插值。
% =========================================================================

%% ========================================================================
%  (Local Function 1) 核心系数矩阵: Nxishujuzhen (精确版)
%  对应 Canuto & Taira (2015) 及张绍龙《三维NS》附录公式
% ========================================================================
function [Gam, At, Bt, Dt, V1, V2, V12] = Nxishujuzhen(p, m, Par)
    
    % --- 0. 解包物理量 ---
    rho = p.rho; u = p.u; v = p.v; T = p.T; 
    mu = p.mu; dmudT = p.dmudT; d2mudT2 = p.d2mudT2;
    
    ux = p.ux; uy = p.uy; vx = p.vx; vy = p.vy;
    rhox = p.rhox; rhoy = p.rhoy;
    mux = p.mux; muy = p.muy;
    
    % 基本流二阶导数 (用于精确计算源项 D)
    uxx=p.uxx; uyy=p.uyy; uxy=p.uxy;
    vxx=p.vxx; vyy=p.vyy; vxy=p.vxy;
    
    % [核心修正] 接收 Part 2 传入的温度二阶导数
    % 如果 Part 2 未传递这些值，此处会报错。
    % 物理意义: 用于计算基本流的热传导源项 div(k' grad T)
    Txx = p.Txx; 
    Tyy = p.Tyy;
    
    Re = Par.Re; invRe = 1/Re;
    Pr = Par.Pr; gM2 = Par.gamma * Par.Ma^2;
    Cv = Par.Cv;
    
    % 应变率张量分量 & 散度
    s_xx = ux; s_yy = vy; s_xy = 0.5*(uy+vx);
    div_u = ux + vy;

    % --- 1. 矩阵 Gamma (时间项系数) ---
    % 对应方程左端: Gamma * dq/dt
    Gam = diag([1, rho, rho, rho*Cv]);

    % --- 2. 矩阵 A (Flux Jacobian - X方向) ---
    % 对应对流项线性化: d(F)/dq
    A = zeros(4,4);
    A(1,1) = u; A(1,2) = rho;
    A(2,1) = T/gM2;
    A(2,2) = rho*u - invRe*(4/3*mux);
    A(2,3) = -invRe*muy;
    A(2,4) = rho/gM2 - invRe*(2*s_xx - 2/3*div_u)*dmudT; 
    
    A(3,2) = (2/3)*invRe*muy;
    A(3,3) = rho*u - invRe*mux;
    A(3,4) = -invRe*2*s_xy*dmudT; 
    
    A(4,2) = rho*T/gM2 - 2*invRe*mu*(2*s_xx - 2/3*div_u);
    A(4,3) = -4*invRe*mu*s_xy;
    A(4,4) = Cv*rho*u - 2*(Par.gamma/(Re*Pr))*mux;
    
    % --- 3. 矩阵 B (Flux Jacobian - Y方向) ---
    % 对应对流项线性化: d(G)/dq
    B = zeros(4,4);
    B(1,1) = v; B(1,3) = rho;
    
    B(2,2) = rho*v - invRe*muy;
    B(2,3) = (2/3)*invRe*mux;
    B(2,4) = -invRe*2*s_xy*dmudT; 
    
    B(3,1) = T/gM2;
    B(3,2) = -invRe*mux;
    B(3,3) = rho*v - (4/3)*invRe*muy;
    B(3,4) = rho/gM2 - invRe*(2*s_yy - 2/3*div_u)*dmudT; 
    
    B(4,2) = -4*invRe*mu*s_xy;
    B(4,3) = rho*T/gM2 - 2*invRe*mu*(2*s_yy - 2/3*div_u);
    B(4,4) = Cv*rho*v - 2*(Par.gamma/(Re*Pr))*muy;

    % --- 4. 矩阵 D (Source Terms - 源项) ---
    % 对应由于基本流梯度产生的源项，包含粘性应力和热传导的线性化
    D = zeros(4,4);
    
    % 辅助系数 (源自张量展开)
    % a_x = d(tau_xx)/dT, a_y = d(tau_yy)/dT
    a_x = (2*s_xx - 2/3*div_u)*p.Tx + 2*s_xy*p.Ty;
    a_y = (2*s_yy - 2/3*div_u)*p.Ty + 2*s_xy*p.Tx;
    
    % b_x, b_y 需要基本流二阶导数 uxx, uyy...
    b_x = (4/3)*uxx + uyy + (1/3)*vxy;
    b_y = (1/3)*uxy + vxx + (4/3)*vyy;
    
    % Row 1 (Continuity)
    D(1,1) = div_u; D(1,2) = rhox; D(1,3) = rhoy;
    
    % Row 2 (X-Momentum)
    D(2,1) = u*ux + v*uy + p.Tx/gM2;
    D(2,2) = rho*ux;
    D(2,3) = rho*uy;
    D(2,4) = rhox/gM2 - invRe*(a_x*d2mudT2 + b_x*dmudT);

    % Row 3 (Y-Momentum)
    D(3,1) = u*vx + v*vy + p.Ty/gM2;
    D(3,2) = rho*vx;
    D(3,3) = rho*vy;
    D(3,4) = rhoy/gM2 - invRe*(a_y*d2mudT2 + b_y*dmudT);

    % Row 4 (Energy)
    D(4,1) = Cv*(u*p.Tx + v*p.Ty) + T/gM2*div_u;
    D(4,2) = Cv*rho*p.Tx;
    D(4,3) = Cv*rho*p.Ty;
    
    % [完全精度修正] 能量方程源项 D(4,4)
    % 项 1: div(k' grad T) -> 线性化展开为: div(dmu/dT * T' * grad T)
    %       = d2mu/dT2 * |grad T|^2 * T' + dmu/dT * lapT * T' + ...
    % 项 2: 粘性耗散 Phi' -> dmu/dT * Phi_structure * T'
    
    gradT_sq = p.Tx^2 + p.Ty^2;
    lapT     = Txx + Tyy; % [NO APPROXIMATION] 使用精确计算的拉普拉斯项
    
    Phi_div_mu = 2 * (s_xx^2 + s_yy^2 + 2*s_xy^2 - (1/3)*div_u^2);
    
    D(4,4) = (rho/gM2)*div_u ...
             - (Par.gamma/(Re*Pr)) * (gradT_sq * d2mudT2 + lapT * dmudT) ... 
             - invRe * Phi_div_mu * dmudT;

    % --- 5. 粘性扩散矩阵 V (Viscous Diffusion) ---
    % 对应二阶导数项的系数
    Vxx = zeros(4,4); Vyy = zeros(4,4); Vxy = zeros(4,4);
    c_mu = invRe * mu;
    
    Vxx(2,2) = (4/3)*c_mu; Vxx(3,3) = c_mu; Vxx(4,4) = (Par.gamma/Pr)*c_mu;
    Vyy(2,2) = c_mu; Vyy(3,3) = (4/3)*c_mu; Vyy(4,4) = (Par.gamma/Pr)*c_mu;
    Vxy(2,3) = (1/3)*c_mu; Vxy(3,2) = (1/3)*c_mu;
    
    % --- 6. 坐标变换 (Metric Transformation) ---
    % 将物理空间的矩阵映射到计算空间 (xi, eta)
    Curv_Xi  = m.xi_xx*Vxx + m.xi_yy*Vyy + 2*m.xi_xy*Vxy;
    Curv_Eta = m.eta_xx*Vxx + m.eta_yy*Vyy + 2*m.eta_xy*Vxy;
    
    At = m.xi_x*A + m.xi_y*B - Curv_Xi;
    Bt = m.eta_x*A + m.eta_y*B - Curv_Eta;
    Dt = D;
    
    V1  = m.xi_x^2*Vxx + m.xi_y^2*Vyy + 2*m.xi_x*m.xi_y*Vxy;
    V2  = m.eta_x^2*Vxx + m.eta_y^2*Vyy + 2*m.eta_x*m.eta_y*Vxy;
    V12 = 2*(m.xi_x*m.eta_x*Vxx + m.xi_y*m.eta_y*Vyy) + ...
          (m.xi_x*m.eta_y + m.xi_y*m.eta_x)*Vxy;
end

%% ========================================================================
%  (Local Function 2) 高阶差分算子生成: get_fd_stencil
%  功能: 生成 4 阶中心差分及边界偏心差分矩阵
% ========================================================================
function [d1, d2] = get_fd_stencil(N, type, h)
    e = ones(N,1);
    
    % --- 1. 内部点 (Central 5-Point, 4th Order) ---
    % 一阶导数
    d1 = spdiags([e -8*e 0*e 8*e -e], -2:2, N, N)/(12*h);
    % 二阶导数
    d2 = spdiags([-e 16*e -30*e 16*e -e], -2:2, N, N)/(12*h^2);
    
    if strcmp(type, 'periodic')
        % --- 周期性边界 (Periodic) ---
        % 处理矩阵角点 (Wrap around)
        d1(1,N-1:N)=[1 -8]/(12*h);   d1(N,1:2)=[8 -1]/(12*h);
        d1(2,N)=1/(12*h);            d1(N-1,1)=-1/(12*h);
        
        d2(1,N-1:N)=[-1 16]/(12*h^2); d2(N,1:2)=[16 -1]/(12*h^2);
        d2(2,N)=-1/(12*h^2);          d2(N-1,1)=-1/(12*h^2);
    else
        % --- 非周期边界 (Non-periodic) ---
        % 必须使用偏心差分 (Biased Stencils) 以防止越界，且保持 4 阶精度
        
        % D1 Forward (Row 1, 2)
        d1(1,1:5) = [-25, 48, -36, 16, -3] / (12*h);
        d1(2,1:5) = [-3, -10, 18, -6, 1] / (12*h);
        
        % D1 Backward (Row N, N-1)
        d1(N,N-4:N) = [3, -16, 36, -48, 25] / (12*h);
        d1(N-1,N-4:N) = [-1, 6, -18, 10, 3] / (12*h);
        
        % D2 Forward (Standard 4th order biased)
        d2(1,1:5) = [45, -154, 214, -156, 61] / (12*h^2);
        d2(2,1:5) = [11, -20, 6, 4, -1] / (12*h^2); % Adjusted coefficients
        
        % D2 Backward
        d2(N,N-4:N) = [11, -56, 114, -104, 35] / (12*h^2); % Adjusted coefficients
        d2(N-1,N-4:N) = [-1, 4, 6, -20, 11] / (12*h^2);
    end
end

%% ========================================================================
%  (Local Function 3) 原始数据读取: Read_Raw_Data
%  关键功能: 直接读取，不插值，处理周期性重合点
% ========================================================================
function [X, Y, Q, Cfg] = Read_Raw_Data(Cfg, Par)
    
    fprintf('   [IO] 正在读取文件 %s (Expected Size: %dx%d)...\n', ...
            Cfg.filename, Cfg.Nx, Cfg.Ny);
    
    try
        raw = load(Cfg.filename);
    catch
        error('无法加载文件。请确保文件存在且为纯文本格式(去除表头)。');
    end
    
    Ni = Cfg.Nx; % 200
    Nj = Cfg.Ny; % 477 (假设包含重合点)
    
    % --- 1. 维度检查 ---
    % 检查总行数是否匹配 Ni*Nj
    if size(raw, 1) ~= Ni*Nj
        warning('文件行数 (%d) 与预设 Nx*Ny (%d) 不匹配！尝试自动推断...', ...
                size(raw,1), Ni*Nj);
        Nj = size(raw,1) / Ni;
        if floor(Nj) ~= Nj
            error('无法整除 Ni=%d，请检查网格尺寸设置。', Ni);
        end
        fprintf('   -> 自动修正 Nj = %d\n', Nj);
    end

    % --- 2. 恢复数据结构 (Reshape) ---
    X0   = reshape(raw(:,1), Ni, Nj);
    Y0   = reshape(raw(:,2), Ni, Nj);
    % 请注意核对您的 Tecplot 数据列顺序！
    % 假设: 5=U, 6=V, 9=T
    U0   = reshape(raw(:,5), Ni, Nj);
    V0   = reshape(raw(:,6), Ni, Nj); 
    T0   = reshape(raw(:,9), Ni, Nj);
    
    % 密度列检查 (Col 3 vs Col 4)
    % 有些文件格式第3列是密度，有些是第4列(第3列是R)
    TempCol3 = reshape(raw(:,3), Ni, Nj);
    TempCol4 = reshape(raw(:,4), Ni, Nj);
    
    % 简单判断: 密度通常在 1.0 附近，而半径 R 可能很大
    if mean(TempCol4(:)) > 5 && mean(TempCol3(:)) < 2
         Rho0 = TempCol3;
         fprintf('   -> 识别第3列为密度 (Rho)。\n');
    else
         Rho0 = TempCol4;
         fprintf('   -> 识别第4列为密度 (Rho)。\n');
    end
    
    % --- 3. 周期性重合点处理 (Periodic Overlap Removal) ---
    % 检查物理坐标首尾是否重合
    dist_overlap = norm(X0(:,1) - X0(:,end)) + norm(Y0(:,1) - Y0(:,end));
    
    if dist_overlap < 1e-4
        fprintf('   [Periodic] 检测到首尾重合，正在移除最后一列...\n');
        X0(:,end) = []; Y0(:,end) = [];
        U0(:,end) = []; V0(:,end) = [];
        T0(:,end) = []; Rho0(:,end) = [];
        
        Nj = Nj - 1; % 更新维度
        Cfg.Ny = Nj; % 更新全局配置
    else
        fprintf('   [Periodic] 首尾未重合，保留所有数据。\n');
    end

    % --- 4. 打包输出 ---
    X = X0; Y = Y0;
    Q.u = U0; Q.v = V0; Q.T = T0; Q.rho = Rho0;
    
    % 顺便计算 mu (用于后续 Part 1 使用)
    C_mu = 110.4 / Par.T_inf;
    Q.mu = Q.T.^1.5 .* (1+C_mu)./(Q.T+C_mu);
    
    fprintf('   [IO] 数据读取完成。最终网格维度: %d x %d\n', Ni, Nj);
end
